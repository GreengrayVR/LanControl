cmake_minimum_required(VERSION 3.24)
project(LanControl VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "LanControl targets Windows 10/11 x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(LANCONTROL_VERSION_CODE 1 CACHE STRING "Monotonically increasing client release number")
option(LANCONTROL_BUILD_SERVER "Build the Qt Widgets server" ON)
option(LANCONTROL_SINGLE_FILE_CLIENT "Build a static client with embedded enrollment" OFF)
set(LANCONTROL_CLIENT_CONFIG "" CACHE FILEPATH "Shared client.json to embed in the standalone client")
if(LANCONTROL_SINGLE_FILE_CLIENT)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  if(NOT EXISTS "${LANCONTROL_CLIENT_CONFIG}")
    message(FATAL_ERROR "LANCONTROL_CLIENT_CONFIG must point to the shared client.json")
  endif()
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network)
add_library(lan_common STATIC src/common.cpp)
target_include_directories(lan_common PUBLIC src)
target_link_libraries(lan_common PUBLIC Qt6::Core Qt6::Network shell32 ole32)
target_compile_definitions(lan_common PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX LANCONTROL_VERSION_CODE=${LANCONTROL_VERSION_CODE})
if(LANCONTROL_BUILD_SERVER)
  find_package(Qt6 REQUIRED COMPONENTS Widgets)
  qt_add_executable(LanServer WIN32 src/server.cpp)
  target_link_libraries(LanServer PRIVATE lan_common Qt6::Widgets)
endif()
qt_add_executable(LanClient WIN32 src/client.cpp src/windows.cpp)
target_link_libraries(LanClient PRIVATE lan_common advapi32 shell32 user32 userenv wtsapi32)
if(LANCONTROL_SINGLE_FILE_CLIENT)
  get_target_property(qt_core_type Qt6::Core TYPE)
  if(NOT qt_core_type STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "Single-file client requires a static Qt build")
  endif()
  target_compile_definitions(LanClient PRIVATE LANCONTROL_SINGLE_FILE_CLIENT)
  set_source_files_properties("${LANCONTROL_CLIENT_CONFIG}" PROPERTIES QT_RESOURCE_ALIAS "client.json")
  qt_add_resources(LanClient client_enrollment PREFIX "/lancontrol" FILES "${LANCONTROL_CLIENT_CONFIG}")
  qt_import_plugins(LanClient INCLUDE Qt6::QSchannelBackendPlugin)
endif()
if(MSVC)
  foreach(t lan_common LanClient LanServer)
    if(TARGET ${t})
      target_compile_options(${t} PRIVATE /W4 /utf-8 /permissive-)
    endif()
  endforeach()
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(protocol_tests tests/protocol_tests.cpp)
  target_link_libraries(protocol_tests PRIVATE lan_common Qt6::Test)
  add_test(NAME protocol COMMAND protocol_tests)
  if(LANCONTROL_SINGLE_FILE_CLIENT)
    add_test(NAME standalone_client COMMAND LanClient --self-test)
    qt_add_executable(tls_probe tests/tls_probe.cpp)
    target_link_libraries(tls_probe PRIVATE lan_common)
    qt_add_resources(tls_probe probe_enrollment PREFIX "/lancontrol" FILES "${LANCONTROL_CLIENT_CONFIG}")
    qt_import_plugins(tls_probe INCLUDE Qt6::QSchannelBackendPlugin)
  endif()
endif()
